% README
% This file includes all the neccessary setup for the correct typesetting of the document
% DO NOT CHANGE ANYTHING HERE

% PACKAGES
% Headings
\usepackage{titlesec} % to define heading styles

% Header and footer
\usepackage{fancyhdr} % for the styling of header and footer
\pagestyle{fancy}
\fancyhf{}

% Table of contents, list of figures, list of tables
\usepackage[nottoc,numbib]{tocbibind} % to include the list of figures in the toc
\usepackage{tabularx} % for tables
\usepackage[titles]{tocloft} % titles option is needed such that toc, lof, lot use heading 1 style

% Bullet points
\usepackage{enumitem}

% Figures in LaTeX
\usepackage{graphicx}


% References
\usepackage{calc} % to calculate the indent

% SETUP

% Headings
% Make Heading 1 upper-case and 14pt
\titleformat{\section}
  {\normalfont\bfseries\fontsize{14pt}{16.8pt}\selectfont}
  {\thesection}{1em}{\MakeUppercase}
  
% Make Heading 2 not bold and 12pt
\titleformat{\subsection}
  {\normalfont\fontsize{12pt}{14.4pt}\selectfont}
  {\thesubsection}{1em}{}

% Make Heading 3 not bold and 11pt
\titleformat{\subsubsection}
  {\normalfont\fontsize{11pt}{13.2pt}\selectfont}
  {\thesubsubsection}{1em}{}

% Header & Footer
% Remove header rule
\renewcommand{\headrulewidth}{0pt}

% Header + footer
\fancyhead[R]{\fontsize{9pt}{11pt}\selectfont [Short title of the work]}
\fancyfoot[R]{\fontsize{8pt}{11pt}\selectfont \thepage}

% Table of contents, list of figures, list of tables
% Remove dots in toc
\renewcommand{\cftdot}{}

% increase line spacing in toc. This is difficult to get exactly to 1.5 because of paragraph spacing interactions
\renewcommand{\cftbeforesecskip}{1\baselineskip}
\renewcommand{\cftbeforesubsecskip}{0.75\baselineskip}
\renewcommand{\cftbeforesubsubsecskip}{0.75\baselineskip}

% set the indents for the different levels
% --- Level 1 (section) ---
% Number indent = 0 cm
% Title indent = 1.0 cm
\setlength{\cftsecindent}{0cm}
\setlength{\cftsecnumwidth}{1cm}   % pushes title start to 1 cm

% --- Level 2 (subsection) ---
% Number indent = 1.3 cm
% Title start = 2.5 cm total → numwidth = 2.5 - 1.3 = 1.2 cm
\setlength{\cftsubsecindent}{1.3cm}
\setlength{\cftsubsecnumwidth}{1.2cm}

% --- Level 3 (subsubsection) ---
% Same geometry as Level 2
\setlength{\cftsubsubsecindent}{1.3cm}
\setlength{\cftsubsubsecnumwidth}{1.2cm}

% make Heading 1 levels in toc bold
% --- Level 1 (section) formatting ---
\renewcommand{\cftsecfont}{\bfseries}
\renewcommand{\cftsecpagefont}{\bfseries}
  
% make list of figures numbered
\renewcommand{\listoffigures}{\begingroup
\tocsection
\tocfile{\listfigurename}{lof}
\endgroup}

% make list of tables numbered
\renewcommand{\listoftables}{\begingroup
\tocsection
\tocfile{\listtablename}{lot}
\endgroup}

% Bullet points styling
\setlist[itemize,1]{label = --}   

\setlist[itemize]{
  leftmargin = 1.3cm,
  itemsep = 1.5\baselineskip,  % 1.5 line spacing between items
  labelsep = 0.5cm,
  topsep = 0pt,  % optional: spacing before list
  parsep = 0pt
}

% References
% --- APA hanging indent: exactly 1.27 cm ---
\renewenvironment{CSLReferences}[2]
{
  \begin{list}{}
    {
      \setlength{\leftmargin}{1.27cm}
      \setlength{\itemindent}{-1.27cm}
      \setlength{\itemsep}{#2\baselineskip}
      \setlength{\parsep}{0pt}
    }
}
{\end{list}}
